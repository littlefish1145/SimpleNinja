function(add_engine library directory)
  set(sources)
  foreach(source IN LISTS ENGINE_SOURCES)
    list(APPEND sources "${directory}/${source}")
  endforeach()
  set_source_files_properties("${directory}/getopt.c" PROPERTIES LANGUAGE CXX)
  add_library(${library} STATIC ${sources})
  target_include_directories(${library} PUBLIC "${directory}")
  if(SNINJA_PCH)
    target_precompile_headers(${library} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/engine/precompiled.h")
  endif()
  if(MSVC)
    target_compile_options(${library} PRIVATE /GR-)
  else()
    target_compile_options(${library} PRIVATE -fno-rtti -fno-exceptions)
  endif()
endfunction()

if(WIN32)
  foreach(binary sninja)
    if(TARGET ${binary})
      if(MSVC)
        target_sources(${binary} PRIVATE cmake/ninja.manifest)
      elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_link_options(${binary} PRIVATE
          "LINKER:/MANIFEST:EMBED"
          "LINKER:/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/cmake/ninja.manifest")
      endif()
      add_custom_command(TARGET ${binary} POST_BUILD
        COMMAND icacls $<TARGET_FILE:${binary}> /setintegritylevel M VERBATIM)
    endif()
  endforeach()
endif()

if(BUILD_TESTING)
  find_package(GTest QUIET)
  if(NOT GTest_FOUND)
    include(FetchContent)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest
      URL https://github.com/google/googletest/releases/download/v1.16.0/googletest-1.16.0.tar.gz
      URL_HASH SHA256=78c676fc63881529bf97bf9d45948d905a66833fbfa5318ea2cd7478cb98f399
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(googletest)
  endif()
  find_package(Threads REQUIRED)
  file(GLOB upstream_tests CONFIGURE_DEPENDS src/engine/*_test.cc)
  if(NOT WIN32)
    list(FILTER upstream_tests EXCLUDE REGEX "(includes_normalize|msvc_helper)_test.cc$")
  endif()
  file(GLOB optimization_tests CONFIGURE_DEPENDS tests/engine/*.cc)
  add_executable(sninja_test ${upstream_tests} ${optimization_tests} src/engine/test.cc)
  if(SNINJA_PCH)
    target_precompile_headers(sninja_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/engine/precompiled.h")
  endif()
  target_link_libraries(sninja_test PRIVATE sninja_core GTest::gtest Threads::Threads)
  if(WIN32)
    add_custom_command(TARGET sninja_test POST_BUILD
      COMMAND icacls $<TARGET_FILE:sninja_test> /setintegritylevel M VERBATIM)
  endif()
  add_test(NAME upstream COMMAND sninja_test)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-tmp")
  set_tests_properties(upstream PROPERTIES ENVIRONMENT
    "TMP=${CMAKE_BINARY_DIR}/test-tmp;TEMP=${CMAKE_BINARY_DIR}/test-tmp;TMPDIR=${CMAKE_BINARY_DIR}/test-tmp"
    TIMEOUT 120)


endif()
